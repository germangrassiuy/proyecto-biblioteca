#include <Adafruit_LiquidCrystal.h>

Adafruit_LiquidCrystal lcd(0);

void setup() {
  Serial.begin(9600);

  pinMode(5, OUTPUT);
  pinMode(6, OUTPUT);

  lcd.begin(16, 2);
  lcd.print("Libreria lista");
}

void loop() {

  if (Serial.available()) {

    String texto  = Serial.readStringUntil('|');
    String precio = Serial.readStringUntil('|');
    String existe = Serial.readStringUntil('\n');

    lcd.clear();
    lcd.setCursor(0, 0);
    lcd.print(texto);
    lcd.setCursor(0, 1);
    lcd.print("$" + precio);

    if (existe == "1") {
      digitalWrite(5, HIGH);
      digitalWrite(6, LOW);
      tone(8, 2000, 90);

    } else {
      digitalWrite(5, LOW);
      digitalWrite(6, HIGH);
      tone(8, 400, 400);
    }
  }
}
